## 1.字符串操作

1. `strcpy`

   ```c
   //思路：先保存首地址，只要没遍历完原字符串就一直拷贝。
   char * mystrcpy(char *dest, const char *src)
   {
       char *p = dest; 
       while (*src != '\0')
       {
           *p = *src;
           p++;
           src++;
       }
       *p = '\0';
       return dest;
   }
   char * mystrcpy_v2(char* dest ,const char *src)
   {
       char *p = dest;       // 保存目标首地址
       while((*p++ = *src++)); // 一行搞定拷贝 + 移动 + 结束判断
       return dest;          // 返回目标地址
   }
   ```

2. `strcat`

   ```c
   //思路：先保存首地址，指向dest的尾巴，一直拷贝
   char * mystrcat(char * dest ,const char *src)
   {   
       char *p = dest;
       while (*p)p++;
       while (*p++ = *src++);
       return dest;
   }
   
   /**
    * 功能：字符串连接 (String Concatenate)
    * 将 src 指向的字符串追加到 dest 指向的字符串末尾
    * 参数：
    *   dest: 目标字符串指针 (必须有足够的空间容纳新字符串)
    *   src:  源字符串指针 (只读)
    * 返回值：
    *   返回 dest 的原始指针
    */
   char * mystrcat(char * dest, const char *src)
   {   
       // 1. 保存 dest 的起始地址
       // 因为后面 p 指针会移动，为了最后能返回原始地址，需要一个副本
       char *p = dest;
       
       // 2. 寻找 dest 的末尾
       // 循环直到 *p 为 '\0' (即 0)
       // 循环结束后，p 指向 dest 字符串的结束符 '\0' 的位置
       while (*p)
       {
           p++;
       }
       
       // 3. 复制 src 到 dest 末尾
       // 这是一个经典的 C 语言惯用写法
       // 逻辑分解：
       //   a. *src: 取源字符
       //   b. *p = ... : 赋值给目标位置 (覆盖了 dest 原来的 '\0')
       //   c. 表达式 (*p = *src) 的值就是被赋的值
       //   d. 如果赋的值是 '\0' (0)，while 条件为假，循环结束
       //   e. p++ 和 src++: 指针后移
       while (*p++ = *src++);
       
       // 4. 返回原始目标指针
       // 注意：不能返回 p，因为 p 已经指向了新字符串的末尾
       return dest;
   }
   ```

3. `strcmp`

   ```c
   //思路：字符不等于'\0',只要等于就进循环往后走，不等不进循环作差
   int mystrcmp(const char * str1 ,const char *str2)
   {
       while (*str1 && (*str1 == *str2)
   	{ 
         str1++;
         str2++;
       }
       return (unsigned char)*str1 - (unsigned char)*str2 ;
   }
              
   /**
    * 功能：字符串比较 (String Compare)
    * 比较两个字符串 str1 和 str2 的大小
    * 参数：
    *   str1: 字符串 1
    *   str2: 字符串 2
    * 返回值：
    *   0  : 相等
    *   >0 : str1 大于 str2
    *   <0 : str1 小于 str2
    */
   int mystrcmp(const char * str1, const char *str2)
   {
       // 1. 遍历比较
       // 条件 1: *str1 (确保 str1 没结束)
       // 条件 2: *str1 == *str2 (确保当前字符相等)
       // 只要字符相等且没到末尾，就继续向后移动
       while (*str1 && (*str1 == *str2))
       {
           str1++;
           str2++;
       }
       
       // 2. 返回差值
       // 循环结束时，要么遇到了不同字符，要么遇到了 '\0'
       // 强制转换为 unsigned char 是为了符合 C 标准，避免有符号字符的负数问题
       return (unsigned char)*str1 - (unsigned char)*str2 ;
   }
   ```

4. `memcpy`

   ```c
   //思路：先类型转换，按照（n--）一直赋值。
   void * mymemcpy(void *dest,void const *src,size_t n)
   {
       char *d = (char*)dest;
       char const *s = (char const*)src;
       while (n--)
       {
           *d++ = *s++;
       }
       return dest;
   }
   
   /**
    * 功能：将内存块 src 的前 n 个字节复制到内存块 dest  ❗(不适用于内存重叠的情况)
    * 参数：
    *   dest: 目标内存区域的指针 (可写)
    *   src:  源内存区域的指针 (只读)
    *   n:    要复制的字节数
    * 返回值：
    *   返回目标指针 dest 的原始值 (便于链式调用或检查)
    */
   void * mymemcpy(void * dest, void const *src, size_t n)
   {
       // 1. 类型转换
       // void * 指针不能直接进行解引用 (*) 或自增 (++) 操作
       // 将其转换为 char *，因为 char 的大小保证为 1 字节，适合按字节复制
       char *d = (char*)dest;
       
       // 源指针也转换为 char *，并加上 const 限定，防止意外修改源数据
       char const *s = (char const*)src;
   
       // 2. 循环控制
       // n--: 先使用 n 的值判断循环 (当 n 为 0 时停止)，然后 n 自减
       // 这确保循环恰好执行 n 次
       while (n--)
       {
           // 3. 核心复制逻辑
           // *s: 取出源指针当前指向的字节
           // *p = ... : 将该字节赋值给目标指针当前指向的位置
           // s++ : 源指针向后移动 1 个字节
           // p++ : 目标指针向后移动 1 个字节
           // 注意：这里是后置自增，先取值赋值，再移动指针
           *d++ = *s++;
       }
   
       // 4. 返回原始目标指针
       // 此时 p 已经指向了 dest + n 的位置，所以不能返回 p
       // 必须返回函数参数中的原始 dest 地址，符合标准库 memcpy 的行为
       return dest;
   }
   ```

5. memmove

   ```c
   //思路：多考虑一种内存重叠的情况
   void *mymemmove(void *dest, const void *src, size_t n)
   {
       char *d = (char *)dest;
       const char *s = (const char *)src;
       // 有重叠，从后往前拷贝
       if (d > s && d < s + n)
       {
           // 指针指向最后一个字节
           d += n - 1;
           s += n - 1;
           while (n--)
           {
               *d-- = *s--;
           }
       }
       // 无重叠，正常从前到后拷贝
       else
       {
           while (n--)
           {
               *d++ = *s++;
           }
       }
       return dest;
   }
   
   /**
    * 功能：安全内存拷贝 (支持内存重叠)
    * 将源内存区域 src 的前 n 个字节拷贝到目标区域 dest
    * 参数：
    *   dest: 目标内存指针 (可写)
    *   src:  源内存指针 (只读)
    *   n:    要拷贝的字节数
    * 返回值：
    *   返回 dest 的原始指针
    */
   void *mymemmove(void *dest, const void *src, size_t n)
   {
       // 1. 类型转换
       // 转换为 char * 以便按字节操作，同时保留源指针的 const 属性
       char *d = (char *)dest;
       const char *s = (const char *)src;
   
       // 2. 重叠检测与方向决策
       // 判断条件：d > s && d < s + n
       // 含义：目标起始地址 在 [源起始地址, 源结束地址) 区间内
       // 场景：目标区域在源区域"后面"且两者有重叠
       // 示例：源[100~199], 目标[150~249], 重叠区间[150~199]
       if (d > s && d < s + n)
       {
           // === 情况A：有重叠，且目标在源之后 ===
           // 策略：从后往前拷贝 (Backward Copy)
           // 原因：如果从前向后，源数据会被新数据覆盖，导致后续拷贝错误
           
           // 指针移动到最后一个字节的位置
           d += n - 1;  // d 指向 dest + n - 1
           s += n - 1;  // s 指向 src + n - 1
   
           // 从尾到头逐字节拷贝
           while (n--)
           {
               *d-- = *s--;  // 先赋值，然后指针向前移动
           }
       }
       // 3. 无重叠 或 目标在源之前
       else
       {
           // === 情况B：无重叠 或 目标在源之前 ===
           // 策略：从前向后拷贝 (Forward Copy) - 与 memcpy 相同
           // 即使有重叠，如果目标在源之前 (d < s)，从前向后拷贝也是安全的
           // 示例：源[150~249], 目标[100~199], 重叠[150~199]
           // 先拷贝前面的字节不会破坏后面还没读的源数据
           
           while (n--)
           {
               *d++ = *s++;  // 先赋值，然后指针向后移动
           }
       }
       // 4. 返回原始目标指针
       return dest;
   }
   ```
