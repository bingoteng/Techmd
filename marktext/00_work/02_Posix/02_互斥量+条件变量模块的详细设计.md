```mermaid
erDiagram
    GlobalResources ||--|| 静态资源static : "持有全局同步原语"

    pthread_mutex_t ||--|| s_srh_mutex : "mutex_semid持有(1:1)"
    pthread_mutex_t ||--|| pthread_mutexattr_t : "mutex_attr内嵌(1:1)"
    pthread_mutex_t ||--|| s_srh_spinlock : "lock内嵌(1:1)"

    s_srh_mutex ||--o| s_srh_thread : "owner指向(0..1 当前持有者)"

    pthread_mutex_t {
        int32_t mutex_valid "第1成员: 互斥锁有效性标识"
        int32_t init_done "第2成员: 自动初始化状态机"
        s_srh_mutex_ptr mutex_semid "第3成员: 内核互斥锁对象指针"
        pthread_mutexattr_t mutex_attr "第4成员: 互斥锁属性对象"
        s_srh_spinlock lock "第5成员: 保护内部状态的自旋锁"
        uint8_t ref_cnt "第6成员: 操作引用计数器"
    }

    pthread_mutexattr_t {
        int mutex_attr_status "第1成员: 属性初始化状态"
        int mutex_attr_protocol "第2成员: 优先级反转协议"
        int mutex_attr_prioceiling "第3成员: 优先级天花板值"
        int mutex_attr_type "第4成员: 互斥锁类型"
    }

    s_srh_mutex {
        s_srh_spinlock lock "保护互斥锁内核对象"
        s_srh_thread_ptr owner "当前持有锁的TCB"
        s_srh_dlist_node wait_list "阻塞等待线程链表"
    }

    s_srh_thread {
        void_ptr data "反向指向internalPthread"
        int32_t prio "内核优先级"
    }

    静态资源static {
        s_srh_spinlock _module_lock "模块级别的初始化同步锁"
        pthread_mutexattr_t default_mutex_attr "全局默认互斥锁属性"
    }
```
