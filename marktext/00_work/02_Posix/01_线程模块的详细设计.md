```mermaid
erDiagram
    GlobalResources ||--o{ 线程内部信息结构体internalPthread: "管理活跃线程列表"
    GlobalResources ||--|| 静态资源static : "持有全局同步原语"

    线程内部信息结构体internalPthread ||--|| s_srh_thread : "tcb指向(1:1 运行时)"
    线程内部信息结构体internalPthread }o--o| cleanup_handler : "handlerBase串联(栈式链表)"
    线程内部信息结构体internalPthread ||--|| s_srh_spinlock : "state_lock内嵌(1:1)"

    s_srh_thread ||--|| 线程内部信息结构体internalPthread: "data指向(1:1 反向关联)"
    s_srh_thread ||--|| sched_param_t : "sched内嵌(1:1)"

    线程内部信息结构体internalPthread {
        s_srh_dlist_node list_node "串联全局线程管理链表"
        s_srh_thread_ptr tcb "指向内核线程控制块"
        int64_t flags "状态标志位(VALID/JOINABLE等)"
        srh_sem_ptr joiner_wait "阻塞pthread_join的信号量"
        s_srh_spinlock state_lock "保护内部状态（attribute）的自旋锁"
        sched_param_t priority "POSIX调度参数(优先级/CPU)"
        thread_start_args start_arg "入口函数与参数"
        void_ptr exit_status "线程退出码/返回值"
        cleanup_handler_ptr handlerBase "清理函数栈顶指针"
        pid_t id "线程所属进程标识符"
    }

    s_srh_thread {
        void_ptr data "反向指向internalPthread"
        sched_param_t sched "内核调度参数"
        void_ptr entry "内核线程入口函数"
        void_ptr entry_arg "入口函数参数"
        char name "线程名"
        uint32_t policy "调度策略"
        uint32_t base_prio "内核基础优先级"
        uint32_t default_cpu_id "默认CPU亲和性"
    }

    cleanup_handler {
        void_ptr routine "清理回调函数"
        void_ptr arg "传给清理函数的参数"
        cleanup_handler_ptr next "下一个清理处理程序"
    }

    静态资源static {
        s_srh_dlist_node g_pthread_list_head "活跃线程链表头"
        s_srh_spinlock g_pthread_list_lock "全局线程链表锁"
        pthread_attr_t defaultPthreadAttr "默认线程属性"
        int pid_cntr "线程ID计数器"
    }

    pthread_attr_t {
        int threadAttrStatus "属性对象状态（如已初始化/未初始化）"
        size_t threadAttrStacksize "栈大小（字节）"
        void_ptr threadAttrStackaddr "栈起始地址"
         char_ptr threadAttrName "线程名称"
        int threadAttrDetachstate "分离属性 （JOINABLE/DETACHED）"
        int threadAttrSchedpolicy "内核调度策略（RR/FIFO）"
        sched_param_t threadAttrSchedparam "调度参数（优先级）"
        int threadAttrInheritsched "调度继承模式（INHERIT/EXPLICIT）"
    }

    sched_param_t {
        uint8_t prio "POSIX实时优先级(1-60)"
        uint32_t cpu "CPU亲和性编号"
    }

 
```
