##### 4.实现1-wire单总线通信协议（LSB优先）

###### 1.复位

```c
static int ds18b20_reset(void)
{
    int timeout；

// 1.主机拉低480~960us（复位脉冲）
    gpio_set_value(ds18b20gpio, 0);
    udelay(500);  // 500us 在范围内

//2.切换为输入，检测从机拉低（存在脉冲）
    gpio_direction_input(ds18b20gpio);

//3.从机在15-60us内拉低总线，timeout超时处理，最大等待时间300*1us 覆盖检测完整存在脉冲的时间  
    timeout = 300；
    while (gpio_get_value(ds18b20gpio) && timeout--) 
    udelay(1);
    if (timeout <= 0) return -1;  // 无响应
// 4.等待从机释放总线（变高）
    timeout = 300；
    while (!gpio_get_value(ds18b20gpio) && timeout--); 
    if (timeout <= 0) return -1;
//5.等待恢复期
    udelay(500);  
    gpio_direction_output(ds18b20gpio, 1);  // 恢复输出模式

    return 0;
}

⚠️主从异步，复位是唯一同步机制
```

###### 2.写时序

```c
static void ds18b20_writebit(unsigned char data)
{
    gpio_set_value(ds18b20gpio, 0);  // 拉低总线
    udelay(5);                        // 保持 >1us（满足最小1us要求）

    if (data)
     gpio_set_value(ds18b20gpio, 1); // 若写1，则提前释放总线

    udelay(75);                      // 总时隙至少60us，这里115+5=120us
    gpio_set_value(ds18b20gpio, 1);   // 无论写0/1，最后都拉高（保险）
    udelay(5);                        // 短暂延时，稳定总线
}

static void ds18b20_writebyte(unsigned char byte)
{
    int i = 0;
    for (i = 0; i < 8; i++)
    {
        ds18b20_writebit(byte & 0x01); // 从最低位开始发送
        byte >>= 1;
    }
}
```

###### 3.读时序

```c
static unsigned char ds18b20_readbit(void)
{
    unsigned char data = 0;

    gpio_set_value(ds18b20gpio, 0);     // 主机拉低
    udelay(5);                          // >1us（最小要求）
    gpio_set_value(ds18b20gpio, 1);     // 释放总线
    gpio_direction_input(ds18b20gpio);  // 切换为输入（高阻态，靠上拉电阻）
    udelay(10);                          // 等待 10us（采样点应在15us内）

    data = gpio_get_value(ds18b20gpio); // 读取当前电平

    gpio_direction_output(ds18b20gpio, 1); // 切回输出，拉高
    udelay(50);                            // 等待剩余时间，总时隙(5+10+60)>60us

    return data;
}
static unsigned char ds18b20_readbyte(void)
{
    unsigned char byte = 0;
    int i = 0;

    for (i = 0; i < 8; i++)
    {
        if (ds18b20_readbit())
            byte |= 0x1 << i;   // LSB 优先，第i位是第i次读到的bit
    }

    return byte;
}
```

###### 4.读取温度

```c
static int ds18b20_readvalue(unsigned long *pvalue)
{
    int ret = 0;
    unsigned short th = 0;
    unsigned short tl = 0;

    ret = ds18b20_reset();   // ① 复位 + 检测存在脉冲
    if (-1 == ret) return -1;

    ds18b20_writebyte(0xcc); // ② Skip ROM（单设备时用）
    ds18b20_writebyte(0x44); // ③ Convert T（启动温度转换）

    msleep(500);             // ④ 等待转换完成（12位精度最大750ms，500ms对9~12位安全）

    ret = ds18b20_reset();   // ⑤ 再次复位
    if (-1 == ret) return -1;

    ds18b20_writebyte(0xcc); // ⑥ Skip ROM
    ds18b20_writebyte(0xbe); // ⑦ Read Scratchpad（读暂存器）

    tl = ds18b20_readbyte(); // ⑧ 读低字节
    th = ds18b20_readbyte(); // ⑨ 读高字节
    *pvalue = ((th << 8) | tl); // ⑩ 组合成16位有符号整数（温度原始值）

    return 0;
}

工作流程：复位 -> 跳过ROM(CCH) -> 启动温度转换(44H) -> 延时等待 -> 复位 -> 跳过ROM(CCH) ->读取温度(BEH) -> 接收温度数据 
复位：单总线，通过复位操作，实现同步机制。
```

##### 2、DS18B20的时序简单描述一下？

`DS18B20 `基于单总线（1-Wire）协议，通信时序严格依赖微秒级精准控制，包含三个核心阶段：

① **复位时序** —— 主机拉低总线`480~960μs`后释放总线（发出一个复位脉冲），DS18B20检测到IO引脚上升沿后等待`15-60us`，拉低总线`60~240μs`返回一个低电平信号的存在脉冲；

```
主拉低-释放-从拉低
```

② **写时序** —— 主机写“0”需拉低`>60μs`，写“1”拉低`1~15μs`后释放总线；

```
主写0    主拉低60us 
主写1    主拉低1-15us释放
```

③ **读时序** —— 主机拉低总线`1-15μs`后释放总线，`DS18B20`在`15μs`内控制总线,若发送0，继续拉低总线；若发送1，则释放总线。主机必须在`10-15us`内采样总线电平。

```
主读0 主拉低-释放-从15us内拉低-主采样
主读1    主拉低-释放-从释放-主采样
```

—— 一句话总结：**“拉低复位 → 按位读写（`LSB先行`）→ 等转换 → 读高低字节”，全程微秒控时，缺一不可。**

#### 3、`ADXL345`核心函数：

##### 1、初始化传感器adxl345_init()

```c
/* ADXL345 寄存器定义 */
#define ADXL345_DEVID          0x00  // 设备ID寄存器
#define ADXL345_DEVID_VAL      0xE5   // 预期设备ID值
#define ADXL345_DATA_FORMAT    0x31  // 数据格式寄存器
#define ADXL345_POWER_CTL      0x2D  // 电源控制寄存器
#define ADXL345_DATAX0         0x32  // X轴数据起始地址

//读时序0000 0000 bit7：1读（0x80）/0写 ，bit6：1多字节（0x40）/0单字节，bit5-bit0 寄存器地址

int adxl345_init(struct spi_device *spi)
{
    int ret;
    u8 tx[2], rx[2];

    /* 验证设备ID */
    tx[0] = ADXL345_DEVID | 0x80;  // 读命令1000 0000
    ret = spi_write_then_read(spi, tx, 1, rx, 1);
    if (ret < 0 || rx[0] != ADXL345_DEVID_VAL) {
        pr_info("Device ID check failed (got 0x%02X, expected 0x%02X)\n", rx[0], ADXL345_DEVID_VAL);
        return -ENODEV;
    }

    /* 配置数据格式 (±16g, 全分辨率) */
    tx[0] = ADXL345_DATA_FORMAT;  // 寄存器地址0x31
    tx[1] = 0x0B;                 // 0x0B = 1011 (全分辨率 + ±16g)[6](@ref)
    ret = spi_write_then_read(spi, tx, 2, NULL, 0);
    if (ret < 0) 
    return ret;

    /* 启动测量模式 */
    tx[0] = ADXL345_POWER_CTL;      //0x2D
    tx[1] = 0x08;                 // 0x08 = 测量模式[6](@ref)

    return spi_write_then_read(spi, tx, 2, NULL, 0);
}

关键：spi_write_then_read函数（）
1.验证设备id
2.配置数据格式±16g，满量程测量
3.启动测量
```

##### 2、读三轴加速度功能函数adxl345_read_xyz()

```c
/* 读取三轴加速度的主函数 */
int adxl345_read_xyz(struct spi_device *spi, int *x, int *y, int *z)
{
    int ret;
    u8 tx_buf[7] = {0};  // 发送缓冲区：命令+6字节数据
    u8 rx_buf[7] = {0};  // 接收缓冲区

 struct spi_transfer tr = {
    .tx_buf = tx_buf,
    .rx_buf = rx_buf,
    .len = 7,// 1字节命令 + 6字节数据，发出1个命令字节，连续读取6个数据字节，其他字节无效/或为任意值
        .delay_usecs = 20,        //通信间隔20us通信一个字节
    };

 /* 构造读取命令 (0x80: 读标志 | 0x40: 多字节模式) */
    tx_buf[0] = ADXL345_DATAX0 | 0x80 | 0x40;  // 连续读取6字节数据 
 /* 构造spi消息，由链表管理*/
    struct spi_message msg;
    spi_message_init(&msg);
    spi_message_add_tail(&tr, &msg);

    /* 执行SPI传输 */
    ret = spi_sync(spi, &msg);//同步传输，阻塞等待
    if (ret < 0) {
        pr_err("SPI transfer failed: %d\n", ret);
        return ret;
    }

    /* 解析原始数据 (小端模式: LSB在前) */
    *x = (s16)((rx_buf[2] << 8) | rx_buf[1]); 
    *y = (s16)((rx_buf[4] << 8) | rx_buf[3]);
    *z = (s16)((rx_buf[6] << 8) | rx_buf[5]);

    return 0;
}
关键函数：spi_transfer tr/spi_message msg
        spi_message_init（）
        spi_message_add_tail（）
         spi_sync(spi, &msg)//同步传输，阻塞等待
```

#### 1、`I2C`子系统驱动框架介绍：

​    `I2C`子系统是驱动分层思想的典型体现它主要分为4层结构，包括:

1、`物理硬件层`：具体的`i2c`硬件设备

​    包括 `SoC `内置的` I2C 控制器`（Master）和外部` I2C 从设备`（如传感器、`EEPROM` 等 Slave 设备）通过`SCL`、`SDA`进行连接通信；

4、`i2c设备驱动层`：自己实现的`i2c_client`设备和`i2c_driver`驱动。

5、`用户空间`：通过open，read，write，close调用驱动层实现的操作函数

### 2、`SPI`子系统：

#### 1、`SPI`子系统介绍：

`SPI子系统`和`I2C子系统`都是根据驱动分层思想设计的，但是硬件通信协议等决定了它们性能，应用场景等不同。

4. **设备驱动层**：由 `spi_device`（代表具体 `SPI 从设备`）和 `spi_driver`（对应设备的驱动程序）组成，我主要完成设备驱动层的开发实现设备初始化、数据交互及用户空间接口等功能。

这一分层设计使得 `SPI 驱动`具有良好的可移植性和可扩展性，上层驱动无需关心底层控制器细节，底层控制器驱动也无需耦合具体外设逻辑，充分体现了 Linux 内核驱动分层思想。

#### 3、AD转换原理：`采样-比较-保留`

```c
ADC转换原理详细解释
以12位为分辨率，3.3v基准电压为例。
1.采用逐次逼近法，SAR寄存器将一个模拟电压转换为`0-4095`的数字量。
2.首先将最高位（bit11）设为1其余位为0，（由SAR寄存器控制）生成对应的试探电压（2048*3.3/4096=1.65v）
3.然后将输入电压和该试探电压比较，若输入电压大于等于该试探电压：保留该位为1，否则该位清0。
4.从最高位到最低位重复此过程一共比较12次，得到一个12位二进制数换算为10进制数再乘以3.3/4096就能的到原始输出电压。
```

```
    "SAR ADC 的核心是'二分查找'思想。采样保持电路先锁定输入电压，然后 SAR 逻辑从最高位开始，每次用 DAC 生成试探电压，比较器决定该位保留还是清零。12 位分辨率需要 12 次比较，加上采样时间，总转换时间约几十微秒
```

## 六、`MQTT`协议:

### 1、`MQTT`协议简介：

​        `MQTT`->中文名："消息队列遥测传输协议"，是一种基于**发布/订阅**模式下的**轻量级**通信协议，专为低带宽、高延迟、不可靠网络环境下的物联网设备设计。端口号（默认1883，加密8883）

### 2、`MQTT和HTTP`详细对比表格：

| 特性            | `MQTT`                     | `HTTP`                  |
|:------------- |:-------------------------- |:----------------------- |
| ✅**通信模式**     | **发布/订阅**                  | **请求-响应**               |
| ✅**设计目标**     | 轻量级、低带宽、低功耗、高实时性           | 统一、通用、可读性强、功能丰富         |
| **协议基础**      | **二进制协议**                  | **文本协议**                |
| ✅**报文头开销**    | **非常小**（最小2字节）             | **很大**（包含大量文本头信息）       |
| ✅**连接方式**     | **默认长连接**                  | **默认短连接**（HTTP/1.1可长连接） |
| **数据传输方向**    | **双向通信**（客户端可随时发/收）        | **单向通信**（必须先由客户端发起请求）   |
| **QoS（服务质量）** | **原生支持 3 个级别**（QoS 0,1,2）  | **不支持**，可靠性依赖底层TCP      |
| **心跳机制**      | **原生支持**（PINGREQ/PINGRESP） | **不支持**，需通过应用层或TCP实现    |
| **适用场景**      | **物联网、移动推送、实时消息**          | **Web服务、文件传输**          |

⬇️表格分点详细解释:

#### 1、通信模式：发布/订阅 vs.请求-响应

- **MQTT（发布/订阅）**：
  - **工作方式**：消息的发送者（**发布者Publish**）和接收者（**订阅者Subscribe**）通过一个中间人（**代理人Broker**）解耦。发布者将消息发送到某个**主题**，代理负责将其转发给所有订阅了该主题的订阅者。
  - **优势**：
    - **一对多通信**：一条消息可以轻松分发给无数个接收者。
    - **解耦**：发布者不知道也不关心有多少订阅者，订阅者也不知道谁是发布者。
    - **实时推送**：数据一旦产生，立即被推送给所有感兴趣的客户端。
- **HTTP（请求-响应）**：
  - **工作方式**：客户端主动向服务器发起一个**请求**，服务器处理请求后返回一个**响应**。通信总是由客户端发起。
  - **劣势**：
    - **一对一通信**：一个请求对应一个响应。
    - **紧耦合**：服务器必须等待客户端的请求才能发送数据。
    - **实时性差**：为了获取新数据，客户端必须不断地向服务器**轮询**，产生大量无效请求。

#### 2、协议开销：二进制 vs. 文本

这是导致两者在**带宽和功耗**上差异巨大的核心原因。

- **MQTT（二进制）**：
  - 协议头被精心设计为用最少的比特位表示。一个最简单的 PUBLISH 消息头可能只有 **2 字节**。
  - **就像发电报**，每个字符都要计费，所以用最精简的代码表达意思。
- **HTTP（文本）**：
  - 协议头是人类可读的文本，包含 `GET`, `POST`, `200 OK`, `Content-Type` 等大量字符串。
  - 一个简单的 HTTP 请求头轻松超过 **50-100 字节**，是 MQTT 的数十倍。
  - **就像写信**，有固定的格式和客套话，虽然易读，但篇幅长。

#### 3、连接管理：长连接 vs. 短连接

- **MQTT（长连接）**：
  - 客户端与代理建立一次 TCP连接后，会长期保持。所有消息都通过这个连接进行传输。
  - **优势**：避免了频繁建立和断开 TCP 连接的开销（三次握手、TLS握手），极大地降低了延迟和网络负担。
- **HTTP（默认短连接）**：
  - 在 HTTP/1.0 中，每次请求-响应后连接都会关闭。HTTP/1.1 引入了长连接，但它的本质是“保持连接一段时间以供复用”，其设计核心仍是围绕离散的请求-响应周期，而非 MQTT 那样的持续事件驱动模型。

#### 4、数据传输方向：双向 vs. 单向

- **MQTT（双向）**：
  - 一旦连接建立，客户端和代理都可以**随时**向对方发送消息。客户端既可以发布数据，也可以随时接收来自代理的订阅消息。
- **HTTP（单向）**：
  - 通信必须由**客户端发起**。服务器无法主动向客户端推送消息。要实现“服务器推送”效果，需要使用 **`WebSocket`**、**Server-Sent Events** 或客户端**轮询**等变通方案，这些都增加了复杂性。

#### 5、服务质量

- **`MQTT`（原生支持 `QoS`）**：
  - **`QoS 0`**：最多一次（可能丢失）。
  - **`QoS 1`**：至少一次（可能重复）。
  - **`QoS 2`**：确保一次（不丢失不重复）。
  - 这为在不可靠网络上的通信提供了不同级别的可靠性保证。
- **HTTP（依赖 TCP）**：
  - HTTP 本身不提供消息传递的保证级别。它依赖于底层 TCP 的可靠性，即“确保一次”。如果请求中途失败，客户端需要自己处理重试逻辑。

| `01` | 1   | **仅采集温湿度**   | `DHT11`          | 4 字节（温 2+ 湿 2）      |
| `10` | 2   | **温湿度 + 光照** | `DHT11 + BH1750` | 6 字节（温 2+ 湿 2+ 光 2） |
| `11` | 3   | **保留**       | -                | -                   |

#### Byte[1]中采样间隔定义（`bit5-bit0`）

| 采样间隔  | 二进制         | 十进制       | 备注                 |
| ----- |:----------- | --------- | ------------------ |
| 以秒为单位 | `0x00-0x3F` | `0-63s/次` | 对应`bit5-bit0`的十进制数 |

#### Byte[1]中响应回复

| 变量类型      | 设置失败   | 设置成功   |
| --------- | ------ | ------ |
| `uint8_t` | `0x00` | `0x01` |

### 4️⃣数据读取-数据帧格式（`0x02`）

| 状态        | 帧头     | ID     | *命令*     | 长度高    | 长度低                  | *数据*       | 校验和     | 帧尾     | 一包         |
| --------- | ------ | ------ | -------- | ------ | -------------------- | ---------- | ------- | ------ | ---------- |
| `MCU`接收命令 | `0x55` | `0x11` | *`0x02`* | `0x00` | `0x00`               | `ND`       | `Cksum` | `0xAA` | `7B`       |
| 响应命令      | `0x55` | `0x22` | *`0x02`* | `0x00` | **`0x02/0x04/0x06`** | **`根据模式`** | `Cksum` | `0xAA` | `9/11/13B` |

#### 响应数据字节说明（根据采集模式变化）

| 采集模式 | 数据长度 | `Byte[1:2]` | `Byte[3:4]` | `Byte[5:6]` |

| 仅温湿度 | 4 字节 | `温度 (2B)`   | `湿度 (2B)`   | -           |
| 全采集  | 6 字节 | `光照 (2B)`   | `温度 (2B)`   | `湿度 (2B)`   |

### 5️⃣代码实现示例（STM32 HAL 库）

```c
// ================= 系统配置结构体 =================
typedef struct {
    uint8_t collect_mode;      // 采集模式：0=仅光照，1=仅温湿度，2=全采集，3=保留
    uint8_t sample_interval;   // 采样间隔：0-63 秒
    uint8_t collect_light;     // 是否采集光照 (BH1750)
    uint8_t collect_temp;      // 是否采集温度 (DHT11)
    uint8_t collect_humi;      // 是否采集湿度 (DHT11)
} SystemConfig_t;

SystemConfig_t g_system_config = {0};

// ================= 解析参数设置命令 (0x01) =================
/**
 * @brief  解析上位机下发的配置字节 (Byte[1])
 * @param  config_byte: 位域格式 [bit7-6:模式][bit5-0:间隔]
 * @retval 0=成功，-1=失败(保留模式)
 */
int ParseSetConfigCommand(uint8_t config_byte)
{
    // 1. 解析采集模式 (高 2 位: bit7-bit6)
    uint8_t mode = (config_byte >> 6) & 0x03;

    // 根据协议定义解析模式
    switch (mode) {
        case 0: // 00: 仅采集光照强度 (BH1750)
            g_system_config.collect_light = 1;
            g_system_config.collect_temp = 0;
            g_system_config.collect_humi = 0;
            break;

        case 1: // 01: 仅采集温湿度 (DHT11)
            g_system_config.collect_light = 0;
            g_system_config.collect_temp = 1;
            g_system_config.collect_humi = 1;
            break;

        case 2: // 10: 温湿度 + 光照 (全采集)
            g_system_config.collect_light = 1;
            g_system_config.collect_temp = 1;
            g_system_config.collect_humi = 1;
            break;

        case 3: // 11: 保留 (无效模式)
            printf("Warning: Reserved mode received, ignore\n");
            return -1;  // 返回失败，上位机将收到 0x00 响应
    }

    g_system_config.collect_mode = mode;

    // 2. 解析采样间隔 (低 6 位: bit5-bit0)
    uint8_t interval = config_byte & 0x3F;  // 掩码 00111111
    if (interval > 0) {
        g_system_config.sample_interval = interval;  // 单位：秒
    }

    printf("配置成功：模式=%d(0=光/1=温湿/2=全), 间隔=%d 秒\n", 
           mode, g_system_config.sample_interval);
    return 0;
}

// ================= 封装响应数据 (0x02) =================
/**
 * @brief  根据采集模式封装传感器数据
 * @param  data_buffer: 输出缓冲区 (至少 6 字节)
 * @retval 实际数据长度 (2/4/6 字节)，-1=错误
 * 
 * 数据顺序 (协议定义):
 *   全采集 (6 字节): [光照 2B][温度 2B][湿度 2B]
 *   仅温湿度 (4 字节): [温度 2B][湿度 2B]
 *   仅光照 (2 字节): [光照 2B]
 */
int BuildResponseData(uint8_t *data_buffer)
{
    if (data_buffer == NULL) return -1;

    int data_len = 0;

    // ⭐ 关键：按协议顺序封装 - 光照在前，温湿度在后
    // 这样无论哪种模式，上位机解析逻辑统一

    // 1. 光照数据 (BH1750) - 2 字节
    if (g_system_config.collect_light) {
        float light = BH1750_Read();  // 读取光照值 (单位：Lux)

        // 转换为 16 位整数 (大端模式：高字节在前)
        uint16_t light_val = (uint16_t)light;  // 范围 0-65535 Lux
        data_buffer[data_len++] = (light_val >> 8) & 0xFF;  // 高字节
        data_buffer[data_len++] = light_val & 0xFF;          // 低字节
    }

    // 2. 温湿度数据 (DHT11) - 4 字节
    if (g_system_config.collect_temp || g_system_config.collect_humi) {
        float temp = 0, humi = 0;

        // 读取 DHT11 (单次通信获取温湿度)
        if (DHT11_Read(&temp, &humi) != 0) {
            // 读取失败，返回默认值 0
            temp = 0;
            humi = 0;
        }

        // 温度 2 字节：放大 10 倍保留 1 位小数 (大端模式)
        // 例：25.6℃ → 256 → 0x01 0x00
        int16_t temp_val = (int16_t)(temp * 10);
        data_buffer[data_len++] = (temp_val >> 8) & 0xFF;  // 高字节
        data_buffer[data_len++] = temp_val & 0xFF;          // 低字节

        // 湿度 2 字节：放大 10 倍保留 1 位小数 (大端模式)
        // 例：70.0% → 700 → 0x02 0xBC
        int16_t humi_val = (int16_t)(humi * 10);
        data_buffer[data_len++] = (humi_val >> 8) & 0xFF;  // 高字节
        data_buffer[data_len++] = humi_val & 0xFF;          // 低字节
    }

    return data_len;  // 返回 2/4/6
}

// ================= 串口发送完整响应帧 =================
/**
 * @brief  发送数据上报帧 (命令 0x02 响应)
 * @param  mode: 采集模式 (用于计算数据长度)
 * @retval 0=成功，-1=失败
 */
int SendDataResponse(uint8_t mode)
{
    uint8_t tx_buffer[256] = {0};
    int idx = 0;

    // 1. 帧头 + ID+ 命令
    tx_buffer[idx++] = 0x55;  // 帧头
    tx_buffer[idx++] = 0x22;  // MCU 响应，ID=0x22
    tx_buffer[idx++] = 0x02;  // 命令字：数据上报

    // 2. 封装数据并获取长度
    uint8_t data_payload[6] = {0};
    int data_len = BuildResponseData(data_payload);
    if (data_len < 0) return -1;

    // 3. 数据长度 (2 字节，大端模式)
    tx_buffer[idx++] = (data_len >> 8) & 0xFF;  // 高字节 (恒为 0)
    tx_buffer[idx++] = data_len & 0xFF;          // 低字节 (2/4/6)

    // 4. 数据载荷
    memcpy(&tx_buffer[idx], data_payload, data_len);
    idx += data_len;

    // 5. 校验和 (ID+ 命令 + 长度 + 数据)
    uint8_t checksum = 0;
    for (int i = 1; i < idx; i++) {  // 从 ID(索引 1) 开始累加
        checksum += tx_buffer[i];
    }
    tx_buffer[idx++] = checksum;

    // 6. 帧尾
    tx_buffer[idx++] = 0xAA;

    // 7. 发送 (阻塞发送，确保完整)
    HAL_UART_Transmit(&huart1, tx_buffer, idx, 100);

    printf("发送响应：len=%d, mode=%d\n", idx, mode);
    return 0;
}

// ================= 主循环集成示例 =================
void MainLoop(void)
{
    static uint32_t last_sample_time = 0;

    while (1) {
        // 1. 处理串口接收 (状态机解析，见下文)
        UART_ProcessReceivedData();

        // 2. 周期性采集 (根据配置的间隔)
        uint32_t current_time = HAL_GetTick();
        if (current_time - last_sample_time >= g_system_config.sample_interval * 1000) {
            last_sample_time = current_time;

            // 可选：主动上报 (如果协议支持定时自动上报)
            // SendDataResponse(g_system_config.collect_mode);
        }

        // 3. 阈值判断 + 本地控制 (声光报警/风扇)
        CheckThresholdAndControl();

        // 4. 低功耗等待
        HAL_Delay(10);
    }
}
```

### 🔧 配套：串口接收状态机（应用层解析）

```c
// ================= 串口接收状态机 =================
typedef enum {
    STATE_WAIT_HEAD,    // 等待帧头 0x55
    STATE_WAIT_ID,      // 等待 ID
    STATE_WAIT_CMD,     // 等待命令字
    STATE_WAIT_LEN_H,   // 等待长度高字节
    STATE_WAIT_LEN_L,   // 等待长度低字节
    STATE_WAIT_DATA,    // 等待数据 (可变长)
    STATE_WAIT_CHECK,   // 等待校验和
    STATE_WAIT_TAIL,    // 等待帧尾 0xAA
    STATE_ERROR         // 错误状态
} UartParseState_t;

static UartParseState_t parse_state = STATE_WAIT_HEAD;
static uint8_t rx_frame[256];
static uint16_t rx_index = 0;
static uint16_t expected_len = 0;

/**
 * @brief  串口接收中断回调 (HAL 库)
 * @note   只负责字节收集，解析放在主循环
 */
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if (huart->Instance == USART1) {
        uint8_t byte = rx_byte;  // 从中断缓冲区读取
        UART_ParseByte(byte);    // 送入状态机
        HAL_UART_Receive_IT(&huart1, &rx_byte, 1);  // 重新开启中断
    }
}

/**
 * @brief  状态机解析单个字节
 * @param  byte: 接收到的字节
 * @retval 0=帧解析成功，-1=解析中/错误
 */
int UART_ParseByte(uint8_t byte)
{
    switch (parse_state) {
        case STATE_WAIT_HEAD:
            if (byte == 0x55) {
                rx_frame[0] = byte;
                rx_index = 1;
                parse_state = STATE_WAIT_ID;
            }
            break;

        case STATE_WAIT_ID:
            rx_frame[1] = byte;
            // 可选：验证 ID 是否合法 (0x11=PC→MCU, 0x22=MCU→PC)
            parse_state = STATE_WAIT_CMD;
            break;

        case STATE_WAIT_CMD:
            rx_frame[2] = byte;
            parse_state = STATE_WAIT_LEN_H;
            break;

        case STATE_WAIT_LEN_H:
            rx_frame[3] = byte;
            parse_state = STATE_WAIT_LEN_L;
            break;

        case STATE_WAIT_LEN_L:
            rx_frame[4] = byte;
            // 计算期望数据长度 (大端模式)
            expected_len = ((uint16_t)rx_frame[3] << 8) | rx_frame[4];
            rx_index = 5;
            parse_state = (expected_len > 0) ? STATE_WAIT_DATA : STATE_WAIT_CHECK;
            break;

        case STATE_WAIT_DATA:
            rx_frame[rx_index++] = byte;
            if (rx_index >= 5 + expected_len) {
                parse_state = STATE_WAIT_CHECK;
            }
            break;

        case STATE_WAIT_CHECK:
            rx_frame[rx_index++] = byte;  // 保存校验和
            parse_state = STATE_WAIT_TAIL;
            break;

        case STATE_WAIT_TAIL:
            if (byte == 0xAA) {
                // 帧接收完成，验证校验和
                uint8_t calc_sum = 0;
                for (int i = 1; i < rx_index - 1; i++) {  // ID 到数据
                    calc_sum += rx_frame[i];
                }

                if (calc_sum == rx_frame[rx_index - 1]) {
                    // ✅ 校验成功，处理命令
                    ProcessFrame(rx_frame, rx_index);
                } else {
                    printf("Checksum error: expect 0x%02X, got 0x%02X\n", 
                           calc_sum, rx_frame[rx_index - 1]);
                }
            }
            // 无论成功失败，重置状态机
            parse_state = STATE_WAIT_HEAD;
            rx_index = 0;
            break;

        default:
            parse_state = STATE_WAIT_HEAD;  // 错误恢复
            break;
    }
    return (parse_state == STATE_WAIT_HEAD && rx_index == 0) ? 0 : -1;
}

/**
 * @brief  处理完整帧 (应用层业务逻辑)
 * @param  frame: 完整帧数据
 * @param  len: 帧长度
 */
void ProcessFrame(uint8_t *frame, int len)
{
    uint8_t cmd = frame[2];  // 命令字

    switch (cmd) {
        case 0x01:  // 参数设置
            if (len >= 7) {  // 最小帧：头+ID+ 命令 + 长度 (2)+ 数据 (1)+ 校验 + 尾
                uint8_t config_byte = frame[6];  // Byte[1]
                if (ParseSetConfigCommand(config_byte) == 0) {
                    // 设置成功，发送响应
                    SendConfigResponse(0x01);  // 0x01=成功
                } else {
                    // 设置失败 (保留模式)
                    SendConfigResponse(0x00);  // 0x00=失败
                }
            }
            break;

        case 0x02:  // 数据读取
            // 立即响应传感器数据
            SendDataResponse(g_system_config.collect_mode);
            break;

        default:
            printf("Unknown command: 0x%02X\n", cmd);
            break;
    }
}

/**
 * @brief  发送配置响应帧
 * @param  status: 0x00=失败，0x01=成功
 */
void SendConfigResponse(uint8_t status)
{
    uint8_t tx_buffer[9] = {
        0x55, 0x22, 0x01,  // 头+ID+ 命令
        0x00, 0x01,        // 长度=1
        status,            // 数据：0x00/0x01
        0x00, 0xAA         // 校验和 (待计算)+ 帧尾
    };

    // 计算校验和
    tx_buffer[7] = 0x22 + 0x01 + 0x00 + 0x01 + status;

    HAL_UART_Transmit(&huart1, tx_buffer, 9, 50);
}
```

### 7️⃣面试官追问预测

##### *追问 1：为什么选择` 0x55 和 0xAA` 作为帧头帧尾？*

​    ***参考回答**：*

> *"这两个值二进制是 `01010101`和`10101010`，**跳变沿丰富**，便于接收方时钟同步。同时它们在正常数据中出现的概率较低，可以减少**误判帧边界**的情况。如果数据中出现 `0x55`，状态机会正确识别为数据而非帧头。"*

##### *追问 2：如果数据中恰好出现 `0x55 或 0xAA` 怎么办？*

​    ***参考回答**：*

> *"当前协议采用**状态机顺序解析**，只有在帧头状态才会识别` 0x55`，数据状态中的 `0x55` 会被当作普通数据处理。*

##### *追问 3：BH1750光照的一次的原始数据是多少？为什么只传2字节？ DHT11 一次通信要 40bit 数据，你为什么只传 4字节？*

​    ***参考回答**：*

> *"**通信协议数据 ≠ 传感器原始数据**。*
> 
> 1. ***BH1750**：原始数据就是 **16 位（2 字节）**，所以直接传 2 字节，无需压缩。*
> 2. ***DHT11**：原始通信是 **40 位（5 字节）**，但包含 1 字节校验和。MCU 校验后丢弃校验位，只传 **4 字节有效数据**（温湿度各 2 字节）。将温度其合并为**2 字节 uint16 格式**（例如温度 25.6℃ 传为 256 或 2560）湿度也是如此。*
> 
> *这样设计是为了**节省串口带宽**，同时把**校验和解析的复杂度留在 MCU 端**，上位机只需处理有效数据。”*

##### *追问 4：如果上位机下发的采集模式是保留值（11），下位机怎么处理？*

​    ***参考回答**：*

> *"下位机解析后会识别为无效模式，**保持原有配置不变**，并向上位机返回'设置失败'（`0x00`）的响应帧。这样可以防止误操作导致系统异常。"*

##### *追问 5：校验和只有 1 字节，会不会容易冲突？*

​    ***参考回答**：*

> *"确实，1 字节校验和的检错率约 99.6%，对于**短帧**（<50 字节）和**低误码率**的串口通信足够。如果需要更高可靠性，可以用 `CRC16` 校验（2 字节），检错率 99.998%。我的项目传输距离短、波特率低，累加和校验已满足需求。"*

##### *追问 6：如何解决串口粘包问题？*

​    ***参考回答**：*

> *"粘包是因为接收方处理速度慢于发送方。我的解决方案：*
> 
> 1. ***固定帧结构**：通过长度字段明确每帧边界；*
> 2. ***状态机解析**：严格按帧头→长度→数据→帧尾的顺序解析，不会跨帧；*
> 3. ***超时重置**：如果接收超时（如 100ms 未收到下一字节），自动丢弃当前缓冲，重置状态机。"*

##### *追问 7：这个协议和你` i.MX6ULL` 项目的`MQTT `协议有什么区别？*

***参考回答**：*

> *"**场景不同**：*
> 
> * ***`STM32 串口`协议**：是我**自主设计**的轻量级二进制协议，最小 8 字节，适合**有线点对点**（`MCU↔PC`），开销小。*
> * ***`i.MX6ULL MQTT` 协议**：是**标准物联网协议**，基于 `JSON` 文本，开销大但通用性强，适合**长距离物联网**（设备↔云平台），支持主题订阅和 `QoS`。*

##### 追问 8：阈值判断是如何设计的？

```
    在这个项目中，由于 `DHT11`和`BH1750` 都是被动式传感器，无法主动上报越限事件，因此我采用**周期性轮询采集 + 软件阈值判断**的方式具体是通过主循环每 2 秒读取一次传感器数据，然后在应用层比较是否超出预设阈值，若超限则控制 `GPIO `驱动声光报警器。
```

##### 追问 9：模型是什么模型？

```
系统采用类 C/S 架构：Qt 上位机作为客户端（`Modbus `主站），周期性通过串口向 `STM32 `下位机（作为服务端/`Modbus` 从站）发送数据请求；`STM32 `接收请求后，采集传感器数据并按 `Modbus`协议封装响应。
```

##### 追问 9：物理层（UART 配置）和应用层（状态机解析）？

```c

“通信分两层实现：
1.物理层（UART 配置）：
    我在 STM32CubeMX 中配置 USART 为 9600bps、8 数据位、无校验、1 停止位（8N1）。上下位机必须严格一致，否则会出现乱码。开启 USART 接收中断，每收到一个字节触发一次中断。
2.应用层（状态机解析）：
    中断服务函数把收到的字节放入环形缓冲区，主循环或后台任务从缓冲区取字节送入状态机。
    状态机按 (0x55) →ID→命令→长度→数据→校验→帧尾 (0xAA) 的顺序解析。
    长度字段决定了状态机需要等待多少个数据字节，这样能有效解决串口粘包问题。
3.关系：
    UART 配置保证字节传输正确，状态机保证数据帧解析正确。两者缺一不可。”
```

```c
    在2025年2月至3月期间，实验室老师安排的项目，我负责开发了一个校园食堂档口信息查询系统。该项目采用B/S架构（浏览器/服务器架构），目标是为师生提供一个高效、直观的食堂档口信息浏览平台。
```

3. **前端展示**：采用AI设计了HTML三级页面结构——一级为登录/注册/注销界面，二级为搜索页，三级展示档口详情（含图文信息），整体注重用户体验与响应速度。
